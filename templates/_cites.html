{# Citations under one AI sentence, each source and page once (several quotes from one page are joined in the hover text). A document citation opens that document at the page; "claim form" and "description" jump to the Basics card. Quotes are model text, so autoescaped as usual. #} {% macro cites(citations, claim_id, documents) -%} {%- set seen = namespace(keys=[]) -%} {%- for c in citations -%} {%- set key = c.source ~ "|" ~ c.page -%} {%- if key not in seen.keys -%} {%- set seen.keys = seen.keys + [key] -%} {%- set quotes = citations | selectattr("source", "equalto", c.source) | selectattr("page", "equalto", c.page) | map(attribute="quote") | list -%} {%- if c.source in documents %} {{ c.source }}, p. {{ c.page }} {%- else %} {{ c.source }}, p. {{ c.page }} {%- endif -%} {%- endif -%} {%- endfor -%} {%- endmacro %}
